\documentclass{article}
\usepackage{pgfplots}
\usepackage{animate}
\usepackage[margin=2cm]{geometry}

\pgfplotsset{compat=1.18}
\begin{document}

\begin{center}
    \textbf{🎞 波の動きアニメーション}

    \bigskip

    \begin{animateinline}[autoplay,loop]{10} % 10 fps
        \multiframe{20}{i=0+1}{%
            \begin{tikzpicture}
                \begin{axis}[
                        xmin=0, xmax=6.28,
                        ymin=-1.5, ymax=1.5,
                        axis lines=middle,
                        ticks=none,
                        width=10cm, height=4cm
                    ]
                    \addplot[domain=0:6.28, samples=100, blue, thick]
                    {sin(deg(x - \i*0.2))};
                \end{axis}
            \end{tikzpicture}
        }
    \end{animateinline}

\end{center}

\end{document}
