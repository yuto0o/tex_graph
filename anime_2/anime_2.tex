                \draw[->] (-2,0) -- (6,0) node[right] {$x$};
                \draw[->] (0,-1.5) -- (0,2) node[above] {$y$};

                \pgfmathsetmacro{\a}{\i*0.2}
                \draw[thick,domain=0:6,smooth,variable=\x,blue] plot ({\x},{sin(deg(\x-\a))});

                \node at (4.5,1.5) {\small $\sin(x - \alpha)$, $\alpha=\a$};
            \end{tikzpicture}
        }
